const params = new URLSearchParams(location.search);
const country = params.get("country") || "美国";
document.getElementById("pageTitle").textContent = country + " · 商品列表";
document.title = country + " · 商品列表";

const categories = ["全部", "服饰", "饰品", "家具", "宠物用品"];
const tabsEl = document.getElementById("tabs");
const grid = document.getElementById("grid");
const countEl = document.getElementById("count");
let activeCat = "全部";

function renderTabs() {
  tabsEl.innerHTML = categories.map(c =>
    `<button class="tab ${c === activeCat ? "active" : ""}" data-cat="${c}">${c}</button>`
  ).join("");
  tabsEl.querySelectorAll(".tab").forEach(btn => {
    btn.addEventListener("click", () => {
      activeCat = btn.dataset.cat;
      renderTabs();
      render();
    });
  });
}

function render() {
  const list = products.filter(p =>
    p.country === country && (activeCat === "全部" || p.category === activeCat)
  );
  countEl.textContent = `共 ${list.length} 件商品`;

  if (!list.length) {
    grid.innerHTML = '<div class="empty">该品类暂无商品</div>';
    return;
  }

  grid.innerHTML = list.map(p => `
    <div class="card">
      <img src="${p.image}" alt="${p.name}" loading="lazy">
      <div class="info">
        <div class="name">${p.name}</div>
        <div class="price">¥${p.price.toFixed(2)}</div>
        <div class="meta">数量：${p.quantity}</div>
        <div class="meta">品类：${p.category}</div>
        <div class="addr">地址：${p.address}</div>
      </div>
    </div>
  `).join("");
}

renderTabs();
render();
